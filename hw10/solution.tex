  \item Выполните программу на Welterweight Java (см. динамическую семантику~\cite[\S 2.2]{OstlundWrigstad2010}):
  \begin{enumerate}
    \item выпишите начальную конфигурацию;
    \item выпишите последовательность правил динамической семантики (только названия правил, например \textsc{D-Return}, \textsc{D-Assign} и пр.), которые будут применены при вычислении;
    \item выпишите конечную конфигурацию.
  \end{enumerate}

\begin{minted}[frame=single,linenos,fontsize=\footnotesize]{java}
class Thunk extends Object { Object force() { return this; } }
class Bool extends Thunk {
  Object if(Object t, Object f) {
    Object o; Bool b;
    o = this.force(); b = (Bool)o;
    o = b.if(t, f); return o;
  }
}
class True extends Bool { Object if(Object t, Object f) { return t; } }
class False extends Bool { Object if(Object t, Object f) { return f; } }
class Or extends Bool {
  Bool l; Bool r;
  Object force() {
    Bool b; Bool t; Bool r; Object o;
    b = this.l; t = new True(); r = this.r;
    o = b.if(t, r); return o;
  }
}
class Main extends Object {
  Object main() {
    Bool t; Bool f; Bool x; Bool y; Object o;
    t = new True(); f = new False();
    x = new Or(); x.l = t; x.r = f;
    y = new Or(); y.l = x; y.r = x;
    o = y.force(); return o;
  }
}
\end{minted}

  \textbf{Решение.}

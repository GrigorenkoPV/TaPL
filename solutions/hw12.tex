\section*{Тема 12. Теоретическое задание \textcolor{blue}{(выполнил Павел Григоренко)}}%

\begin{enumerate}
  \item Определите, какие из следующих утверждений о специализации верны.
  Если специализация верна, явно выпишите подстановку, которая её задаёт:
  \begin{enumerate}
    \item \code{∀α.∀β.α→β} $\sqsubseteq$ \code{(Nat→Bool)→Bool}
    \item \code{∀α.∀β.(α→β)→α} $\sqsubseteq$ \code{∀α.(α→α)→α}
    \item \code{∀α.∀β.∀γ.β→α→β→γ} $\sqsubseteq$ \code{∀α.β→(α→(α→Bool))→(β→Bool)}
    \item \code{∀α.∀β.∀γ.∀δ.(γ→β)→(γ→δ)→α} $\sqsubseteq$ \code{∀α.∀β.∀γ.∀δ.∀ε.∀ζ.∀η.(ε→ζ)→(ε→η)→(α→β→γ→δ→Bool)}
  \end{enumerate}

  \color{blue}
  \textbf{Решение.}
  \emph{место для решения}
  \color{black}

  \clearpage

  \item Вычислите $\gen(\Gamma, \code{τ})$ для следующих пар контекста $\Gamma$ и типа \code{τ}:
  \begin{enumerate}
    \item $\Gamma = \smallcode{x : α→δ,\; h : ∀β.β→(γ→β)}$, \smallcode{τ = (α→(γ→Nat))→(δ→(Bool→γ))}
    \item $\Gamma = \smallcode{u : ∀α.(α→β)→α,\; v : (β→ζ)→η}$, \smallcode{τ = ((β→(η→Nat))→α)→(ζ→(α→Bool))}
    \item $\Gamma = \smallcode{g : ∀δ.∀β.(δ→β)→(β→α),\; k : (α→θ)→ι}$, \smallcode{τ = ((β→α)→θ)→(ι→β)}
    \item $\Gamma = \smallcode{p : ∀α.(α→β)→α,\; q : ∀β.(β→α)→(α→β),\; r : γ→δ}$, \smallcode{τ = ((β→(γ→Bool))→δ)→(α→(δ→Nat))}
    \item $\Gamma = \smallcode{m : ∀β.(β→α)→β,\; n : ∀β.(β→γ)→(θ→β),\; s : α→θ}$, \smallcode{τ = ((γ→α)→θ)→(β→γ)}
  \end{enumerate}

  \color{blue}
  \textbf{Решение.}
  \emph{место для решения}
  \color{black}

  \clearpage

  \item Постройте дерево вывода типов в системе Хиндли--Милнера (декларативные синтаксические правила типизации~\cite{ClementDespeyrouxKahnDespeyroux1986}) для следующего замкнутого терма:
\begin{stella}
let h = λf.λg.λx. f (g x) in h h h
\end{stella}

  \color{blue}
  \textbf{Решение.}
  \emph{место для решения}
  \color{black}

  \clearpage

  \item Примените алгоритм J~\cite{Milner1978} к следующему замкнутому терму:
\begin{stella}
let not = λx. if x then false else true in
let arezero = λx. if iszero x then (λx. iszero x) else (λx. false) in
let h = λf. λg. λx. f (g x) in
h h h not arezero
\end{stella}
  Выпишите:
  \begin{enumerate}
    \item итоговый тип терма;
    \item список всех свежих типовых переменных, созданных в процессе работы алгоритма;
    \item для каждой свежей переменной:
    \begin{enumerate}
      \item где она была создана (вызов \code{inst} или правило типизации, и для какого подвыражения);
      \item где она была устранена:
      во время унификации (укажите \code{app}-терм) или
      во время обобщения (укажите \code{let}-терм).
    \end{enumerate}
  \end{enumerate}

  \color{blue}
  \textbf{Решение.}
  \emph{место для решения}
  \color{black}
\end{enumerate}
